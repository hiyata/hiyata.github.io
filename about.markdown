          <p>I am a researcher specializing in the applications of <span class="highlight">artificial intelligence</span> in the field of <span class="highlight">virology</span>. My work primarily focuses on developing computational techniques to model virus behavior, genomes, and their molecular interactions. I currently work under <a href="https://www.med.wayne.edu/profile/dx0934">Dr. Phil Pellett</a> at Wayne State University School of Medicine in Detroit.</p>

I believe that through this, we can uncover patterns in virus behavior that can be exploited to improve our response to novel pathogens. I hope that one day we view viruses as useful tools to help cure disease, enhance human lifespan and immunity, and combat climate change.</p>

document.addEventListener('DOMContentLoaded', (event) => {
  // DNA Animation
  const svg = document.getElementById('dna-svg');
  const width = 1000;
  const height = 200;
  const numBases = 20;
  const baseWidth = width / numBases;
  const helixOffset = 40;

  for (let i = 0; i < numBases; i++) {
    const x = i * baseWidth + baseWidth / 2;
    const y1 = height / 2 + helixOffset * Math.sin(i * 2 * Math.PI / numBases + Math.PI);
    const y2 = height / 2 + helixOffset * Math.sin(i * 2 * Math.PI / numBases);

    const circle1 = document.createElementNS('http://www.w3.org/2000/svg', 'circle');
    circle1.setAttribute('cx', x);
    circle1.setAttribute('cy', y1);
    circle1.setAttribute('r', '5');
    circle1.setAttribute('fill', '#1a1a1a');
    svg.appendChild(circle1);

    const circle2 = document.createElementNS('http://www.w3.org/2000/svg', 'circle');
    circle2.setAttribute('cx', x);
    circle2.setAttribute('cy', y2);
    circle2.setAttribute('r', '5');
    circle2.setAttribute('fill', '#c2410c');
    svg.appendChild(circle2);
  }

  anime({
    targets: '#dna-svg circle:nth-child(odd)',
    translateY: function(el) {
      return parseFloat(el.nextElementSibling.getAttribute('cy')) - parseFloat(el.getAttribute('cy'));
    },
    duration: 1000,
    loop: true,
    easing: 'linear',
    delay: function(el, i) {
      return i * 100;
    },
    direction: 'alternate'
  });

  anime({
    targets: '#dna-svg circle:nth-child(even)',
    translateY: function(el) {
      return parseFloat(el.previousElementSibling.getAttribute('cy')) - parseFloat(el.getAttribute('cy'));
    },
    duration: 1000,
    loop: true,
    easing: 'linear',
    delay: function(el, i) {
      return 50 + i * 100;
    },
    direction: 'alternate'
  });

  // Scroll animations
  const animateOnScroll = (targets, translateY = 50) => {
    anime({
      targets: targets,
      opacity: [0, 1],
      translateY: [translateY, 0],
      duration: 1000,
      easing: 'easeOutQuad',
      delay: anime.stagger(100),
      begin: function(anim) {
        document.querySelectorAll(targets).forEach(el => {
          el.style.opacity = '0';
          el.style.transform = `translateY(${translateY}px)`;
        });
      }
    });
  };

  const observerOptions = {
    threshold: 0.1
  };

  const observer = new IntersectionObserver((entries) => {
    entries.forEach(entry => {
      if (entry.isIntersecting) {
        if (entry.target.classList.contains('intro-section')) {
          animateOnScroll('.intro-text, .intro-vision');
        } else if (entry.target.classList.contains('skills-section')) {
          animateOnScroll('.skill-category');
        } else if (entry.target.classList.contains('personal-interests')) {
          animateOnScroll('.interest-item');
        } else if (entry.target.classList.contains('publications')) {
          animateOnScroll('.publication-item', 20);
        }
        observer.unobserve(entry.target);
      }
    });
  }, observerOptions);

  document.querySelectorAll('.intro-section, .skills-section, .personal-interests, .publications').forEach(section => {
    observer.observe(section);
  });

  // Hero section animations
  anime({
    targets: '.profile-image-container',
    scale: [0, 1],
    opacity: [0, 1],
    duration: 1000,
    easing: 'easeOutElastic(1, .8)'
  });

  anime({
    targets: ['.name', '.title'],
    opacity: [0, 1],
    translateY: [50, 0],
    duration: 1000,
    delay: anime.stagger(200),
    easing: 'easeOutQuad'
  });

  anime({
    targets: '.social-icon',
    opacity: [0, 1],
    translateY: [20, 0],
    duration: 800,
    delay: anime.stagger(100),
    easing: 'easeOutQuad'
  });

});
</script>
